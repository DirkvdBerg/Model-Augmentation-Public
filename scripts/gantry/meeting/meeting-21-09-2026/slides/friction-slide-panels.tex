\panel{How the friction is added}

Garcia puts dry friction in the generalised force vector, Eq.~(15), one term per
physical actuator:

\begin{align*}
F_{X_s}&=(F_1+F_2)-\left[c_{\mathrm{c}1}\sign(\dot X_1)+c_{\mathrm{c}2}\sign(\dot X_2)\right]\\
F_{\Theta_s}&=\left\{(F_1-F_2)-\left[c_{\mathrm{c}1}\sign(\dot X_1)-c_{\mathrm{c}2}\sign(\dot X_2)\right]\right\}\tfrac{L_b}{2}\\
F_{Y_s}&=F_y-c_{\mathrm{c}y}\sign(\dot Y)
\end{align*}

\begin{itemize}
\item $M(Y)$, $C$, $K$ unchanged. Friction enters at the force node only, never in
      the Rayleigh dissipation $D$ (Eq.~5), which holds the viscous terms alone.
\item Arguments are $\sign(\dot X_1),\sign(\dot X_2),\sign(\dot Y)$: rail velocities.
      Using $(\dot X,\dot\Theta,\dot Y)$ instead is wrong and looks plausible.
\item $\delta_a$ carries no friction: internal payload DOF, not a rail.
\item Truth only; the baseline stays frictionless.
\end{itemize}

\begin{equation*}
x_{k+1}=\ftruth(x_k,\ueff{}_{,k})
\qquad\text{against}\qquad
x_{b,k+1}=\fbase(v,x_{b,k},u_k)
\end{equation*}

\panel{Coulomb's law at zero velocity}

\begin{equation*}
|v_i|>0:\quad F_i=c_{\mathrm{c}i}\sign(v_i),
\qquad\qquad
v_i=0:\quad |F_i|\le c_{\mathrm{c}i}.
\end{equation*}

\begin{itemize}
\item Garcia identified at constant velocity: slip force only, no breakaway measured.
\item $\sign(0)=0$ says a stage at rest has no friction. Not the law.
\item Hard $\sign$ gives a differential inclusion; its Filippov solution is the stick branch.
\item No new physical parameter.
\end{itemize}

\panel{Garcia's experiment, replicated}

\begin{align*}
\text{$Y$, with $X_1,X_2$ blocked:}&\qquad F_y=c_y\dot Y+c_{\mathrm{c}y}\sign(\dot Y)\\
\text{$X_1,X_2$ synchronised:}&\qquad F_i=c_{\mathrm{g}i}\dot X_i+c_{\mathrm{c}i}\sign(\dot X_i),
\quad i=1,2
\end{align*}

\begin{itemize}
\item Two experiments, each actuator force read separately.
\item Ours: prescribe constant velocity, solve for the holding force, read the intercept per rail.
\item $M$ constant $\Rightarrow$ that force is exactly affine in $v$ per sign branch.
      Slope is viscous and inertial; the intercept is Coulomb.
\end{itemize}
