\section{Baseline: the model we augment \quad (Eq.~2)}

\noindent\textbf{Coordinates and state.}
\begin{equation}
  q = [\,X,\Theta,Y\,]^\top , \qquad
  \xb = [\,X,\Theta,Y,\dot X,\dot\Theta,\dot Y\,]^\top \in \R^6 ,
  \label{eq:baseline-state}
\end{equation}
with $\xb$ the state of the three degree-of-freedom rigid gantry and $Y$ the scheduling
variable.

\medskip\noindent\textbf{Equations of motion.}
\begin{equation}
  M(Y)\,\ddot q + C\,\dot q + K\,q = P^\top u ,
  \label{eq:baseline-eom}
\end{equation}
with $u = [\,F_1,F_2,F_y\,]^\top \in \R^3$ the stage forces and $P$ the force map
(Eq.~\ref{eq:baseline-out}).

\medskip\noindent\textbf{Inertia matrix.}
\begin{equation}
  M(Y) =
  \begin{bmatrix}
    \alpha            & \beta - m_h Y      & 0 \\[2mm]
    \beta - m_h Y     & \gamma + m_h Y^2   & -m_h d \\[2mm]
    0                 & -m_h d             & m_h
  \end{bmatrix},
  \label{eq:baseline-M}
\end{equation}
with the constants
\begin{equation}
  \alpha := m_1{+}m_2{+}m_b{+}m_h , \qquad
  \beta  := \tfrac{(m_1-m_2)L_b}{2} , \qquad
  \gamma := J_b{+}J_h{+}\tfrac{(m_1+m_2)L_b^2}{4}{+}m_h d^2 .
  \label{eq:baseline-consts}
\end{equation}

\medskip\noindent\textbf{Damping and stiffness.}
\begin{equation}
  C =
  \begin{bmatrix}
    c_{g1}{+}c_{g2}             & (c_{g1}{-}c_{g2})\tfrac{L_b}{2}                    & 0 \\[2mm]
    (c_{g1}{-}c_{g2})\tfrac{L_b}{2} & c_{b1}{+}c_{b2}{+}(c_{g1}{+}c_{g2})\tfrac{L_b^2}{4} & 0 \\[2mm]
    0                          & 0                                                  & c_y
  \end{bmatrix},
  \qquad
  K =
  \begin{bmatrix}
    0 & 0            & 0 \\[2mm]
    0 & k_{b1}{+}k_{b2} & 0 \\[2mm]
    0 & 0            & 0
  \end{bmatrix}.
  \label{eq:baseline-CK}
\end{equation}

\medskip\noindent\textbf{Assumption A.}\quad
$M(Y)$ is nonsingular over the operating range, so $M(Y)^{-1}$ exists (positive-definite
inertia for all physically admissible parameters).

\medskip\noindent\textbf{Discrete-time form.}
\begin{equation}
  \xb_{k+1} = \fb(\xb_k, u_k)\ \ (\text{Eq.\ 2a}) , \qquad
  \yh_k = \hb(\xb_k, u_k) = \Cbar\,\xb_k + D_d\,u_k\ \ (\text{Eq.\ 2b}) ,
  \label{eq:baseline-io}
\end{equation}
with $\yh = [\,X_1,X_2,Y\,]^\top$ the stage positions, obtained by RK4 discretization; the
output selects positions through
\begin{equation}
  C_d = \big[\, P^\top \ \big|\ 0_{3\times3} \,\big] , \qquad
  P =
  \begin{bmatrix}
    1 & 1 & 0 \\
    \tfrac{L_b}{2} & -\tfrac{L_b}{2} & 0 \\
    0 & 0 & 1
  \end{bmatrix} , \qquad
  D_d = 0 ,
  \label{eq:baseline-out}
\end{equation}
and $\Cbar$ is $C_d$ after input/output normalization (Section~\ref{sec:norm}).

\section{System: the data-generating truth \quad (Eq.~1)}

\noindent\textbf{Coordinates and state.}
\begin{equation}
  q = [\,X,\Theta,Y,\delta_a\,]^\top , \qquad
  x = [\,q^\top\ \dot q^\top\,]^\top \in \R^8 ,
  \label{eq:sys-state}
\end{equation}
with the truth adding a hidden tuned-mass absorber (mass $m_a$, relative displacement
$\delta_a$) to the baseline.

\medskip\noindent\textbf{Equations of motion.}
\begin{equation}
  M(Y,\delta_a)\,\ddot q + C_4\,\dot q + K_4\,q = \begin{bmatrix} P^\top u \\ 0 \end{bmatrix} ,
  \label{eq:sys-eom}
\end{equation}
with the payload split $m_h = \mhr + m_a$ (rigid part $\mhr$, absorber $m_a$); the baseline of
Section~1 uses the full payload $m_h$, and the zero force row is the unactuated absorber.

\medskip\noindent\textbf{Inertia matrix.}
\begin{equation}
  M(Y,\delta_a) =
  \begin{bmatrix}
    m_t   & s                  & 0            & 0 \\[2mm]
    s     & J                  & -(\mhr{+}m_a)d & -m_a d \\[2mm]
    0     & -(\mhr{+}m_a)d     & \mhr{+}m_a    & m_a \\[2mm]
    0     & -m_a d             & m_a          & m_a
  \end{bmatrix},
  \label{eq:sys-M}
\end{equation}
with
\begin{align}
  m_t &= m_1{+}m_2{+}m_b{+}\mhr{+}m_a , \notag\\
  s   &= \tfrac{(m_1-m_2)L_b}{2} - (\mhr{+}m_a)Y - m_a(L_0{+}\delta_a) , \label{eq:sys-consts}\\
  J   &= J_b{+}J_h{+}\tfrac{(m_1+m_2)L_b^2}{4} + (\mhr{+}m_a)d^2 + \mhr Y^2 + m_a(Y{+}L_0{+}\delta_a)^2 . \notag
\end{align}
\medskip\noindent\textbf{Damping and stiffness.}
\begin{equation}
  C_4 =
  \begin{bmatrix}
    c_{g1}{+}c_{g2}             & (c_{g1}{-}c_{g2})\tfrac{L_b}{2}                    & 0   & 0 \\[2mm]
    (c_{g1}{-}c_{g2})\tfrac{L_b}{2} & c_{b1}{+}c_{b2}{+}(c_{g1}{+}c_{g2})\tfrac{L_b^2}{4} & 0   & 0 \\[2mm]
    0                          & 0                                                  & c_y & 0 \\[2mm]
    0                          & 0                                                  & 0   & c_a
  \end{bmatrix},
  \qquad
  K_4 =
  \begin{bmatrix}
    0 & 0            & 0 & 0 \\[2mm]
    0 & k_{b1}{+}k_{b2} & 0 & 0 \\[2mm]
    0 & 0            & 0 & 0 \\[2mm]
    0 & 0            & 0 & k_a
  \end{bmatrix}.
  \label{eq:sys-CK}
\end{equation}
with the absorber $(c_a,k_a)$ appended to the baseline diagonal.

\medskip\noindent\textbf{Absorber.}
\begin{equation}
  \begin{aligned}
    & m_a = 0.10\,m_h = 1.01 , \quad
      f_a = 150\ \mathrm{Hz} , \quad
      \zeta_a = 0.05 , \quad
      L_0 = 0.10\ \mathrm{m} , \\
    & k_a = m_a(2\pi f_a)^2 , \quad
      c_a = 2\zeta_a\sqrt{k_a m_a} .
  \end{aligned}
  \label{eq:sys-absorber}
\end{equation}

\medskip\noindent\textbf{Discrete-time form.}
\begin{equation}
  x_{k+1} = f(x_k,u_k) , \qquad y_k = h(x_k) \quad (\text{Eq.\ 1}) ,
  \label{eq:sys-io}
\end{equation}
with RK4 discretization; through $M(Y,\delta_a)$ the absorber couples into the $X,\Theta,Y$
accelerations, and the baseline is recovered as $m_a \to 0$.

\section{Augmentation: dynamic parallel \quad (Eq.~3, Table~1)}

The block scheme gives the topology; the equations below give the state-level routing it
abstracts, namely which states the learning component corrects and which it leaves to the
baseline.

\medskip\noindent\textbf{State decomposition.}
\begin{equation}
  \xh = [\xb;\xa] \in \R^8 , \quad
  \xb = [\,X,\Theta,Y,\dot X,\dot\Theta,\dot Y\,]^\top \in \R^6 , \quad
  \xa = [\,\delta_a,\dot\delta_a\,]^\top \in \R^2 ,
  \label{eq:aug-state}
\end{equation}
with $\xb$ the baseline (physical) state, $\xa$ the added states, and $\xh$ the full model
state. The baseline map $\fb$ reads $\xb$; the network $\phi_{\mathrm{aug}}$ reads the full
$\xh$.

\medskip\noindent\textbf{Learning component.}
\begin{equation}
  w = \phi_{\mathrm{aug}}([\xh;u]) \in \R^4 ,
  \label{eq:aug-ann}
\end{equation}
with $\phi_{\mathrm{aug}}$ a zero-initialised $\tanh$ feedforward network; $w$ is the LFR
interconnection output (Eq.~4).

\medskip\noindent\textbf{Dynamic-parallel model (Table~1).}
\begin{align}
  \xb_{k+1} &= \fb(\xb_k,u_k) + \fa , \label{eq:aug-xb}\\
  \xa_{k+1} &= \ga , \label{eq:aug-xa}\\
  \yh_k     &= \hb(\xb_k,u_k) + \ha , \label{eq:aug-y}
\end{align}
with $\fb$ the Section~1 map, $\hb = \Cbar\,\xb_k + D_d u_k$ the baseline output map (Eq.~5),
and $\ha = 0$.

\medskip\noindent\textbf{Routing: what $\fa$ and $\ga$ are.}
The interconnection $S$ places the four network outputs onto named state rows,
\begin{equation}
  \fa =
  \begin{bmatrix} 0 \\ w_1 \\ 0 \\ 0 \\ w_2 \\ 0 \end{bmatrix}
  \begin{array}{l}\scriptstyle X\\\scriptstyle \Theta\\\scriptstyle Y\\\scriptstyle \dot X\\\scriptstyle \dot\Theta\\\scriptstyle \dot Y\end{array}
  \ \in \R^6 ,
  \qquad
  \ga =
  \begin{bmatrix} w_3 \\ w_4 \end{bmatrix}
  \begin{array}{l}\scriptstyle \delta_a\\\scriptstyle \dot\delta_a\end{array}
  \ \in \R^2 ,
  \label{eq:aug-routing}
\end{equation}
with $w_1,w_2$ correcting the $\Theta$ position and velocity (the only physical axis with
stiffness); $X$ and $Y$ are untouched ($K = 0$), and $w_3,w_4$ drive the added states.

\medskip\noindent\textbf{Simulation.}
\begin{equation}
  \xh_{k_i|k_i} = \psi(u_{<k_i},\,y_{<k_i}) \quad (\text{Eq.\ 5c}) ,
  \label{eq:aug-encoder}
\end{equation}
with $\psi$ the encoder, which sets the initial state once per rollout; the recursion is then
rolled out and $\yh$ compared to $y$ in the loss (Eq.~5a).

\section{Normalization \quad (Eq.~9)}
\label{sec:norm}

Inputs and outputs are scaled to zero mean and unit variance ($T_u$, $T_y$). For the state
scaling $T_x$, Eq.~9 prescribes $\sigma_x$ from the baseline model's simulated states. We
instead take
\begin{equation}
  \sigma_x = \operatorname{std}(x_{\mathrm{log}}) , \qquad
  x_{\mathrm{log}} = \big[\, P^{-\top} y ,\ \ \operatorname{diff}(P^{-\top} y)\,f_s \,\big] ,
  \label{eq:norm}
\end{equation}
that is, from finite-differenced logical states of the measured output. The encoder's
linear-model normalization does use baseline-simulated states, so two state-statistics
sources coexist. This is a deviation from Eq.~9 and a candidate baseline-versus-data mismatch,
most likely on the finite-difference velocity scale.

\section{Training objective and target \quad (Eq.~5)}

Encoder, augmentation, and optionally baseline parameters are estimated by minimizing the
truncated multi-step simulation loss
\begin{equation}
  V_{\mathrm{trunc}} = \frac{1}{2N(T{+}1)}\sum_{i=1}^{N}\sum_{\ell=0}^{T-1}
  \big\| \yh_{k_i+\ell\,|\,k_i} - y_{k_i+\ell} \big\|_2^2 .
  \label{eq:train}
\end{equation}
Acceptance target: noiseless first, then reach the output-noise floor
$\sigma_n = \operatorname{rms}(y)\,10^{-\mathrm{SNR}/20}$ on the validation simulation RMS.
