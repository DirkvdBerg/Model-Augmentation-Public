% bracket from x=#1 to x=#2 at height #3, ticks pointing up (#4=1) or down (#4=-1)
\newcommand{\brk}[4]{\draw (#1,{#3+0.14*#4}) -- (#1,#3) -- (#2,#3) -- (#2,{#3+0.14*#4});}

\newcommand{\ps}[1]{%
\ifnum#1=1 \colorlet{cx}{hl}\else\colorlet{cx}{black}\fi
\ifnum#1=2 \colorlet{c2}{hl}\else\colorlet{c2}{black}\fi
\ifnum#1=3 \colorlet{c3}{hl}\else\colorlet{c3}{black}\fi
\ifnum#1>3 \colorlet{cp}{black!28}\else\colorlet{cp}{c2}\fi
\ifnum#1>1 \colorlet{ct}{black!35}\else\colorlet{ct}{black}\fi
\begin{tikzpicture}
  \useasboundingbox (-0.3,-0.3) rectangle (21.2,8.75);
  \fontsize{17}{20}\selectfont
  % ---------- recorded data and the current sample tau ----------
  \draw[black!45, line width=1.4pt] plot coordinates {\datacurve};
  \node[anchor=south west, text=black!55] at (0.6,7.95) {\normalsize recorded $y,\ u$};
  \draw[black!55, dash pattern=on 3pt off 2pt] (10,8.0) -- (10,0.9);
  \node[anchor=south] at (10,7.95) {$\tau$};
  \node[anchor=south, text=black!55] at (6.5,7.95) {\normalsize past};
  \node[anchor=south, text=black!55] at (13.5,7.95) {\normalsize future};
  % ---------- step 1: the encoder (known from 14a) ----------
  \brk{3.0}{10.0}{6.75}{1}
  \node[anchor=east] at (2.9,6.75) {$\chi_\tau$};
  \node[blk, minimum width=1.8cm, minimum height=0.9cm] (psi) at (6.5,5.35) {$\psi$};
  \draw[->] (6.5,6.75) -- (psi.north);
  \draw[->, ct] (psi.east |- 0,5.6) -- (8.4,5.6) node[right, text=ct] {$\tilde x_\tau$};
  \draw[->, cx, line width=1.2pt] (psi.east |- 0,5.1) -- (8.4,5.1) node[right, text=cx] {$\bar x_\tau$};
  % ---------- step 2, S1: predict the coming residual from x-bar alone ----------
  \ifnum#1>1
    \draw[c2, line width=1.6pt] plot coordinates {\rescurve};
    \begin{scope}[c2]\brk{10.0}{17.0}{4.95}{-1}\end{scope}
    \node[anchor=south west, text=c2] at (10.2,5.0) {$e_\tau$: model error, next $H$ samples};
    \node[blk, draw=cp, minimum width=1.2cm, minimum height=0.85cm] (p) at (9.0,3.2) {\color{cp}$p$};
    \draw[->, cp, line width=1.2pt] (9.0,4.85) -- (p.north);
    \draw[->, cp, line width=1.2pt] (p.east) -- (11.2,3.2) -- (11.2,3.55);
    \node[anchor=north west, text=cp] at (9.9,2.95) {\normalsize predict the coming error from $\bar x_\tau$ alone};
    \ifnum#1>3
      \draw[black!45, line width=1.1pt] ($(p.south west)+(-0.15,-0.1)$) -- ($(p.north east)+(0.15,0.1)$);
    \fi
  \fi
  % ---------- step 3, S2: g_aug learns how x-bar evolves ----------
  \ifnum#1>2
  \begin{scope}[c3]
    \draw[dash pattern=on 4pt off 2pt] (3.233,6.47) -- (3.233,6.33) -- (10.233,6.33) -- (10.233,6.47);

  \end{scope}
  \fi
  % ---------- one line per step ----------
  \ifcase#1\or
    \node[anchor=west, text=hl] at (0.0,0.3) {the encoder also gives the added states $\bar x_\tau$, from random weights};

    \node[anchor=west, text=hl] at (0.0,0.3) {S1: $\bar x$ must hold what predicts the model's error, the missing dynamics};

  \fi
